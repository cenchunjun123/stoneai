    FUNCTION_BLOCK ChillerStage
        VAR_INPUT
            rTempIn                  : REAL;  (* 进水温度 *)
            rTempOut                 : REAL;  (* 出水温度 *)
            rSetpoint                : REAL;  (* 温度设定 *)
        END_VAR
        VAR_OUTPUT
            bStageUp                 : BOOL;  (* 加载指令 *)
            bStageDown               : BOOL;  (* 卸载指令 *)
        END_VAR
        VAR
            rDeadband                : REAL := 0.5;  (* 死区 *)
        END_VAR
    
        bStageUp := FALSE;
        bStageDown := FALSE;
        IF rTempOut > rSetpoint + rDeadband THEN
            bStageUp := TRUE;
        ELSIF rTempOut < rSetpoint - rDeadband THEN
            bStageDown := TRUE;
        END_IF;
    END_FUNCTION_BLOCK
